```rust
use crate::broker::Broker;
use crate::portfolio::Portfolio;
use crate::risk::RiskManager;
use crate::tick::tick;
use crate::types::*;

/// Per-step engine for RL training. Unlike BacktestEngine which takes
/// all bars at once, StepEngine advances one bar at a time so Python
/// can observe state and choose actions between bars.
pub struct StepEngine {
    broker: Broker,
    portfolio: Portfolio,
    risk: RiskManager,
    config: EngineConfig,
    bars: Vec<Bar>,
    current_step: usize,
    equity_curve: Vec<(i64, f64)>,
    fills: Vec<Fill>,
    halted_at: Option<i64>,
    prev_equity: f64,
}

/// Observation returned to Python after each step.
#[derive(Debug, Clone)]
pub struct StepObs {
    /// Normalized OHLCV of current bar
    pub open: f64,
    pub high: f64,
    pub low: f64,
    pub close: f64,
    pub volume: f64,
    /// Portfolio state
    pub equity: f64,
    pub cash: f64,
    pub position_qty: f64,
    pub position_entry_price: f64,
    pub unrealized_pnl: f64,
    pub drawdown: f64,
    /// Step metadata
    pub step: usize,
    pub halted: bool,
}

/// Result of a single step.
pub struct StepResult {
    pub obs: StepObs,
    pub reward: f64,
    pub done: bool,
    pub truncated: bool,
}

impl StepEngine {
    pub fn new(config: EngineConfig, bars: Vec<Bar>) -> Result<Self, EngineError> {
        if config.initial_capital <= 0.0 {
            return Err(EngineError::InvalidConfig("initial_capital must be > 0".into()));
        }
        if bars.is_empty() {
            return Err(EngineError::EmptyInput);
        }
        let initial = config.initial_capital;
        Ok(Self {
            broker: Broker::new(config.broker.clone()),
            portfolio: Portfolio::new(config.initial_capital),
            risk: RiskManager::new(config.risk.clone()),
            config,
            bars,
            current_step: 0,
            equity_curve: Vec::new(),
            fills: Vec::new(),
            halted_at: None,
            prev_equity: initial,
        })
    }

    pub fn reset(&mut self) -> StepObs {
        let initial = self.config.initial_capital;
        self.broker = Broker::new(self.config.broker.clone());
        self.portfolio = Portfolio::new(initial);
        self.risk = RiskManager::new(self.config.risk.clone());
        self.current_step = 0;
        self.equity_curve.clear();
        self.fills.clear();
        self.halted_at = None;
        self.prev_equity = initial;
        self.build_obs()
    }

    /// Step the environment with an action.
    /// action: [strength, target_size] both in [0, 1]
    pub fn step(&mut self, strength: f64, target_size: f64) -> StepResult {
        if self.current_step >= self.bars.len() {
            return StepResult {
                obs: self.build_obs(),
                reward: 0.0,
                done: true,
                truncated: true,
            };
        }

        let bar = self.bars[self.current_step].clone();
        let signal = Signal { strength, target_size };

        self.prev_equity = self.portfolio.state.equity;

        tick(
            &bar,
            &signal,
            &mut self.broker,
            &mut self.portfolio,
            &mut self.risk,
            self.config.broker.fill_model,
            &mut self.equity_curve,
            &mut self.fills,
            &mut self.halted_at,
        );

        self.current_step += 1;

        let done = self.current_step >= self.bars.len() || self.portfolio.state.halted;
        let truncated = self.current_step >= self.bars.len() && !self.portfolio.state.halted;

        // Reward: log return of equity
        let current_equity = self.portfolio.state.equity;
        let reward = if self.prev_equity > 0.0 {
            (current_equity / self.prev_equity).ln()
        } else {
            0.0
        };

        StepResult {
            obs: self.build_obs(),
            reward,
            done,
            truncated,
        }
    }

    fn build_obs(&self) -> StepObs {
        let bar = if self.current_step < self.bars.len() {
            &self.bars[self.current_step]
        } else {
            self.bars.last().unwrap()
        };

        let (pos_qty, pos_entry, unrealized) = match &self.portfolio.state.position {
            Some(p) => (p.quantity, p.entry_price, p.unrealized_pnl),
            None => (0.0, 0.0, 0.0),
        };

        StepObs {
            open: bar.open,
            high: bar.high,
            low: bar.low,
            close: bar.close,
            volume: bar.volume,
            equity: self.portfolio.state.equity,
            cash: self.portfolio.state.cash,
            position_qty: pos_qty,
            position_entry_price: pos_entry,
            unrealized_pnl: unrealized,
            drawdown: self.portfolio.current_drawdown(),
            step: self.current_step,
            halted: self.portfolio.state.halted,
        }
    }

    pub fn num_bars(&self) -> usize {
        self.bars.len()
    }
}
```

```rust
use crate::step::StepEngine as RustStepEngine;

#[pyclass(name = "StepEngine")]
pub struct PyStepEngine {
    inner: RustStepEngine,
}

#[pymethods]
impl PyStepEngine {
    #[new]
    fn new(
        initial_capital: f64,
        broker: &Bound<'_, PyDict>,
        risk: &Bound<'_, PyDict>,
        bars_per_year: f64,
        timestamps: PyReadonlyArray1<'_, i64>,
        opens: PyReadonlyArray1<'_, f64>,
        highs: PyReadonlyArray1<'_, f64>,
        lows: PyReadonlyArray1<'_, f64>,
        closes: PyReadonlyArray1<'_, f64>,
        volumes: PyReadonlyArray1<'_, f64>,
    ) -> PyResult<Self> {
        // Parse config (same as PyBacktestEngine)
        let slippage_bps: f64 = broker.get_item("slippage_bps")?.unwrap().extract()?;
        let fee_rate: f64 = broker.get_item("fee_rate")?.unwrap().extract()?;
        let fill_model_str: String = broker.get_item("fill_model")?.unwrap().extract()?;
        let fill_model = match fill_model_str.as_str() {
            "immediate" => FillModel::Immediate,
            "next_bar" => FillModel::NextBar,
            other => return Err(pyo3::exceptions::PyValueError::new_err(
                format!("invalid fill_model: '{}'", other),
            )),
        };
        let max_position_size: f64 = risk.get_item("max_position_size")?.unwrap().extract()?;
        let max_drawdown_halt: f64 = risk.get_item("max_drawdown_halt")?.unwrap().extract()?;
        let signal_threshold: f64 = risk.get_item("signal_threshold")?.unwrap().extract()?;

        let config = EngineConfig {
            initial_capital,
            broker: BrokerConfig { slippage_bps, fee_rate, fill_model },
            risk: RiskConfig { max_position_size, max_drawdown_halt, signal_threshold },
            bars_per_year,
        };

        // Build bars
        let ts = timestamps.as_slice()?;
        let o = opens.as_slice()?;
        let h = highs.as_slice()?;
        let l = lows.as_slice()?;
        let c = closes.as_slice()?;
        let v = volumes.as_slice()?;
        let bars: Vec<Bar> = (0..ts.len()).map(|i| Bar {
            timestamp: ts[i], open: o[i], high: h[i], low: l[i], close: c[i], volume: v[i],
        }).collect();

        let inner = RustStepEngine::new(config, bars)
            .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;

        Ok(Self { inner })
    }

    fn reset<'py>(&mut self, py: Python<'py>) -> PyResult<Bound<'py, PyDict>> {
        let obs = self.inner.reset();
        Self::obs_to_dict(py, &obs)
    }

    fn step<'py>(&mut self, py: Python<'py>, strength: f64, target_size: f64) -> PyResult<Bound<'py, PyDict>> {
        let result = self.inner.step(strength, target_size);
        let dict = PyDict::new(py);
        dict.set_item("obs", Self::obs_to_dict(py, &result.obs)?)?;
        dict.set_item("reward", result.reward)?;
        dict.set_item("done", result.done)?;
        dict.set_item("truncated", result.truncated)?;
        Ok(dict)
    }

    fn num_bars(&self) -> usize {
        self.inner.num_bars()
    }

    fn obs_to_dict<'py>(py: Python<'py>, obs: &crate::step::StepObs) -> PyResult<Bound<'py, PyDict>> {
        let d = PyDict::new(py);
        d.set_item("open", obs.open)?;
        d.set_item("high", obs.high)?;
        d.set_item("low", obs.low)?;
        d.set_item("close", obs.close)?;
        d.set_item("volume", obs.volume)?;
        d.set_item("equity", obs.equity)?;
        d.set_item("cash", obs.cash)?;
        d.set_item("position_qty", obs.position_qty)?;
        d.set_item("position_entry_price", obs.position_entry_price)?;
        d.set_item("unrealized_pnl", obs.unrealized_pnl)?;
        d.set_item("drawdown", obs.drawdown)?;
        d.set_item("step", obs.step)?;
        d.set_item("halted", obs.halted)?;
        Ok(d)
    }
}
```

```python
"""Gymnasium environment wrapping Rust StepEngine."""

from __future__ import annotations

from pathlib import Path

import gymnasium as gym
import numpy as np
import polars as pl
from gymnasium import spaces


class BacktestEnv(gym.Env):
    """Gymnasium environment for RL-based backtesting.

    Observation: [open, high, low, close, volume, equity, cash,
                  position_qty, entry_price, unrealized_pnl, drawdown]
    Action: [strength, target_size] both continuous in [0, 1]
    Reward: log return of equity per step
    """

    metadata = {"render_modes": []}

    def __init__(
        self,
        data_path: str,
        initial_capital: float = 10_000.0,
        slippage_bps: float = 5.0,
        fee_rate: float = 0.001,
        fill_model: str = "immediate",
        max_position_size: float = 0.20,
        max_drawdown_halt: float = 0.50,
        signal_threshold: float = 0.5,
        bars_per_year: float = 8760.0,
    ):
        super().__init__()

        self.data_path = data_path
        self.initial_capital = initial_capital
        self.broker_config = {
            "slippage_bps": slippage_bps,
            "fee_rate": fee_rate,
            "fill_model": fill_model,
        }
        self.risk_config = {
            "max_position_size": max_position_size,
            "max_drawdown_halt": max_drawdown_halt,
            "signal_threshold": signal_threshold,
        }
        self.bars_per_year = bars_per_year

        # Load data
        df = pl.read_parquet(data_path)
        self._timestamps = df["timestamp"].to_numpy().astype(np.int64)
        self._opens = df["open"].to_numpy().astype(np.float64)
        self._highs = df["high"].to_numpy().astype(np.float64)
        self._lows = df["low"].to_numpy().astype(np.float64)
        self._closes = df["close"].to_numpy().astype(np.float64)
        self._volumes = df["volume"].to_numpy().astype(np.float64)

        # Action: [strength, target_size] in [0, 1]
        self.action_space = spaces.Box(low=0.0, high=1.0, shape=(2,), dtype=np.float32)

        # Observation: 11 continuous values
        self.observation_space = spaces.Box(
            low=-np.inf, high=np.inf, shape=(11,), dtype=np.float32,
        )

        self._engine = None

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self._create_engine()
        obs_dict = self._engine.reset()
        return self._dict_to_array(obs_dict), {}

    def step(self, action):
        strength = float(action[0])
        target_size = float(action[1])

        result = self._engine.step(strength, target_size)

        obs = self._dict_to_array(result["obs"])
        reward = float(result["reward"])
        done = bool(result["done"])
        truncated = bool(result["truncated"])

        return obs, reward, done, truncated, {}

    def _create_engine(self):
        from bt.bt_engine import StepEngine

        self._engine = StepEngine(
            initial_capital=self.initial_capital,
            broker=self.broker_config,
            risk=self.risk_config,
            bars_per_year=self.bars_per_year,
            timestamps=self._timestamps,
            opens=self._opens,
            highs=self._highs,
            lows=self._lows,
            closes=self._closes,
            volumes=self._volumes,
        )

    def _dict_to_array(self, obs_dict: dict) -> np.ndarray:
        return np.array([
            obs_dict["open"], obs_dict["high"], obs_dict["low"],
            obs_dict["close"], obs_dict["volume"],
            obs_dict["equity"], obs_dict["cash"],
            obs_dict["position_qty"], obs_dict["position_entry_price"],
            obs_dict["unrealized_pnl"], obs_dict["drawdown"],
        ], dtype=np.float32)
```

## Task 5: PPO training loop

```python
"""PPO training using stable-baselines3."""

from __future__ import annotations

import json
from pathlib import Path

from bt.config import resolve_path


def train_ppo(
    data_path: str,
    output_dir: str = "models/ppo",
    total_timesteps: int = 100_000,
    initial_capital: float = 10_000.0,
    learning_rate: float = 3e-4,
    n_steps: int = 2048,
    batch_size: int = 64,
    n_epochs: int = 10,
    seed: int = 42,
) -> dict:
    """Train a PPO agent on historical data.

    Returns training summary with path to saved model.
    """
    from stable_baselines3 import PPO
    from stable_baselines3.common.env_util import make_vec_env

    from bt.rl.env import BacktestEnv

    resolved_data = str(resolve_path(data_path))
    output = resolve_path(output_dir)
    output.mkdir(parents=True, exist_ok=True)

    env = BacktestEnv(data_path=resolved_data, initial_capital=initial_capital)

    model = PPO(
        "MlpPolicy",
        env,
        learning_rate=learning_rate,
        n_steps=n_steps,
        batch_size=batch_size,
        n_epochs=n_epochs,
        seed=seed,
        verbose=1,
    )

    model.learn(total_timesteps=total_timesteps)

    model_path = output / "ppo_backtest"
    model.save(str(model_path))

    # Save training config
    config = {
        "data_path": data_path,
        "total_timesteps": total_timesteps,
        "initial_capital": initial_capital,
        "learning_rate": learning_rate,
        "n_steps": n_steps,
        "batch_size": batch_size,
        "n_epochs": n_epochs,
        "seed": seed,
        "model_path": str(model_path),
    }
    (output / "train_config.json").write_text(json.dumps(config, indent=2))

    return {"status": "ok", "model_path": str(model_path), "config": config}
```

```python
"""Strategy that loads a trained PPO policy and generates signals."""

from __future__ import annotations

import numpy as np
import polars as pl


class PPOStrategy:
    """Loads a trained PPO model and generates vectorized signals.

    Runs the policy forward on each bar to produce (strength, target_size).
    """

    def __init__(self, model_path: str, initial_capital: float = 10_000.0):
        self.model_path = model_path
        self.initial_capital = initial_capital

    def compute(self, df: pl.DataFrame) -> pl.DataFrame:
        from stable_baselines3 import PPO

        model = PPO.load(self.model_path)

        n = len(df)
        strengths = np.zeros(n)
        target_sizes = np.zeros(n)

        # Simulate forward pass to get observations
        equity = self.initial_capital
        cash = self.initial_capital
        pos_qty = 0.0
        entry_price = 0.0
        unrealized_pnl = 0.0
        peak_equity = equity

        for i in range(n):
            row = df.row(i, named=True)
            drawdown = (peak_equity - equity) / peak_equity if peak_equity > 0 else 0.0

            obs = np.array([
                row["open"], row["high"], row["low"], row["close"], row["volume"],
                equity, cash, pos_qty, entry_price, unrealized_pnl, drawdown,
            ], dtype=np.float32)

            action, _ = model.predict(obs, deterministic=True)
            strengths[i] = float(np.clip(action[0], 0.0, 1.0))
            target_sizes[i] = float(np.clip(action[1], 0.0, 1.0))

            # Approximate portfolio state update for next observation
            close = row["close"]
            if pos_qty > 0:
                unrealized_pnl = (close - entry_price) * pos_qty
                equity = cash + pos_qty * close
            else:
                equity = cash
            peak_equity = max(peak_equity, equity)

        return pl.DataFrame({
            "strength": pl.Series("strength", strengths),
            "target_size": pl.Series("target_size", target_sizes),
        })
```
